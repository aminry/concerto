  ```proto
  syntax = "proto3";
  package concerto.v1;
  import "google/protobuf/empty.proto";
  // Field numbers in this file are FROZEN as of Task 401.5.

  service Maestro {
    rpc SendToMaestro(MaestroMessageRequest) returns (google.protobuf.Empty);
    rpc GetDigest(GetDigestRequest) returns (Digest);
    rpc SetWorkareaVisibility(VisibilityRequest) returns (google.protobuf.Empty);
  }

  // The user's chat input to the Maestro. `attachments` is a V1.0 text-only
  // seam (design/08 R-9: V1.0 text-only; V1.5 diff/commit-URL drag-drop) —
  // frozen now, populated later; empty in V1.0.
  message MaestroMessageRequest {
    string text = 1;
    repeated MaestroAttachment attachments = 2;
  }

  // Attachment seam (R-9). `kind` is a free-form string so the catalog can
  // grow without a wire break (mirrors the `Chip.action` precedent).
  message MaestroAttachment {
    string kind = 1;   // e.g. "diff" | "commit_url" (V1.5)
    string ref = 2;    // opaque reference resolved by the consumer
  }

  message GetDigestRequest {}

  // The digest rendered above the chat composer (design/08 §3.6). `chips`
  // are the next-step chips; mirrors the suggestions.proto `Chip` wire shape
  // (kept local to this package — field numbers locally owned/frozen).
  message Digest {
    string text = 1;
    repeated Chip chips = 2;
    int64 generated_at_ms = 3;   // unix epoch ms (no google.protobuf.Timestamp)
    bool stale = 4;              // R-7: last-good digest shown with a stale badge when inert
  }

  // Mirror of suggestions.proto `Chip` (rule_id/workarea_id/title/priority/
  // created_at_ms/action). Local copy so maestro.proto is self-contained.
  message Chip {
    string rule_id = 1;
    string workarea_id = 2;
    string title = 3;
    int32 priority = 4;
    int64 created_at_ms = 5;
    string action = 6;
  }

  // Per-workarea Maestro visibility (design/08 §3.3 privacy toggle; 413
  // enforces the blanking). FULL = summaries visible; HARD_FACTS_ONLY =
  // `exclude_from_maestro` (name + hard facts only).
  message VisibilityRequest {
    string workarea_id = 1;
    MaestroVisibility visibility = 2;
  }

  enum MaestroVisibility {
    MAESTRO_VISIBILITY_UNSPECIFIED = 0;
    MAESTRO_VISIBILITY_FULL = 1;
    MAESTRO_VISIBILITY_HARD_FACTS_ONLY = 2;
  }
  ```
